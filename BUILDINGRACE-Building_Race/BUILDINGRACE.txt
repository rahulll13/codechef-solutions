# cook your dish here
import sys
def solve():
    A,B,X,Y = map(int, input().split())
    
    ## 
    chef_time = A * Y    
    chefina_time = B * X  
    
    if chef_time < chefina_time:
        print("CHEF")
    elif chef_time > chefina_time:
        print("CHEFINA")
    else:
        print("BOTH")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
