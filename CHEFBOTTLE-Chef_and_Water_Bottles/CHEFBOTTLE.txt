import sys
def solve():
    N,X,K = map(int, input().split())
    
    if X > K:
        print("0")
    else:
        print(min(N,K // X))
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
