# cook your dish here
def solve():
    A,B = map(int, input().split())
    if A > B:
        print(">")
    elif A < B:
        print("<")
    else:
        print("=")
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError,ValueError):
        pass
