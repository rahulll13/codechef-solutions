import sys
def solve():
    A,B = map(int, input().split())
    C,D = map(int, input().split())
    
    if C >= A and D >= B:
        print("POSSIBLE")
    else: ## AS the Score can't be negative
        print("IMPOSSIBLE")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
