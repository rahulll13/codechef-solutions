import sys
def solve():
    X,Y = map(int, input().split())
    
    if 2 * Y > X:
        print("0")
    else:
        print((X) // (2 * Y))
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
