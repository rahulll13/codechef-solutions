import sys
def solve():
    N = int(input())
    S = input().strip()
    
    # Create a translation table for the DNA complement rules
    trans_table = str.maketrans('ATCG', 'TAGC')
    
    # Apply the translation and print the result
    print(S.translate(trans_table))
    
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
