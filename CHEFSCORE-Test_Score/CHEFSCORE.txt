import sys
def solve():
    N,X,Y = map(int, input().split())
    
    if Y % X == 0 and N * X >= Y:
        print("YES")
    else:
        print("NO")
        
if __name__ == "__main__":

        T = int(input())
        for _ in range(T):
            solve()
