import sys
def solve():
    A,B,X,Y = map(int, input().split())
    
    if (A - Y) <= B <= (A + X):
        print("YES")
    else:
        print("NO")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError,ValueError):
