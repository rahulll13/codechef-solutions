## Condition => Sum of opposite Angles = 180

def solve():
    A,B,C,D = map(int,input().split())
    
    if A + C == 180 and B + D == 180:
        print("YES")
    else:
        print("NO")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
