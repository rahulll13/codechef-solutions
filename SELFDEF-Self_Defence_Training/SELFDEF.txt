import sys
def solve():
    N = int(input())
    Ages = list(map(int, input().split()))
    
    cnt = 0
    for i in range(N):
        if Ages[i] >= 10 and Ages[i] <= 60:
            cnt += 1 
    print(cnt)
    
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
