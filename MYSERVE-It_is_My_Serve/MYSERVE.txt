import sys
def solve():
    P,Q = map(int, input().split())
    
    total_serves = P + Q
    turn = total_serves // 2
    
    if turn % 2 == 0:
        print("Alice")
    else:
        print("Bob")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError,ValueError):
        pass
