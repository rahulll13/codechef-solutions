import math

# Read the number of test cases
for _ in range(int(input())):
    A, B, K = map(int, input().split())
    
    # Calculate the absolute distance between Chef and Chefina
    distance = abs(A - B)
    
    # We need to find the ceiling of (distance / K)
    # Using integer arithmetic, ceil(d / k) can be written as (d + k - 1) // k
    steps = (distance + K - 1) // K
    
    print(steps)
