import sys,math
def solve():
    X,Y = map(int, input().split())
    
    if X < Y:
        print("0")
    else:
        print(math.ceil(X // Y))
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError,ValueError):
        pass
