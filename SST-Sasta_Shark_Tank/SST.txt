import sys
def solve():
    A,B = map(int, input().split())
    
    if A * (100 // 10) > B * (100 // 20):
        print("FIRST")
    elif A * (100 // 10) < B * (100 // 20):
        print("SECOND")
    else:
        print("ANY")
        
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
