import sys
def solve():
    N,A,B = map(int, input().split())
    Even_index = N // 2
    Odd_index = N - Even_index
    Total_duration = (A * Even_index) + (B * Odd_index)
    print(Total_duration)
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError,ValueError):
        pass
