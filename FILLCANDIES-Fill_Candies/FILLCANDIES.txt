import sys
def solve():
    N,K,M = map(int, input().split())
    
    Total_capacity = K * M
    bags = (N + Total_capacity - 1) // Total_capacity
    print(bags)
    
if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
