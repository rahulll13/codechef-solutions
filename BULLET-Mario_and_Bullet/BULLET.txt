import sys,math
def solve():
    X,Y,Z  = map(int, input().split())
    
    anss = (Y // X)
    
    if anss < Z:
        print(abs(Z - anss))
    else:
        print("0")

if __name__ == "__main__":
    try:
        T = int(input())
        for _ in range(T):
            solve()
    except(EOFError, ValueError):
        pass
